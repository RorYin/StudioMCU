#include "stm32f446xe.h"

/*
 * SPI client (slave): another SPI master drives SCK/MOSI into this board.
 */
static void spi1_init(void) {
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_SPI1EN;

    GPIOA->MODER &= ~(GPIO_MODER_MODE5_Msk | GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
    GPIOA->MODER |= (GPIO_MODER_MODE5_ALTERNATE | GPIO_MODER_MODE6_ALTERNATE | GPIO_MODER_MODE7_ALTERNATE);
    GPIOA->AFR[0] &= ~(GPIO_AFR_AFSEL5_Msk | GPIO_AFR_AFSEL6_Msk | GPIO_AFR_AFSEL7_Msk);
    GPIOA->AFR[0] |= (GPIO_AFR_AFSEL5_AF5 | GPIO_AFR_AFSEL6_AF5 | GPIO_AFR_AFSEL7_AF5);

    SPI1->CR1 = SPI_CR1_SSM | SPI_CR1_SSI | SPI_CR1_SPE;
}

int main(void) {
    spi1_init();
    while (1) {
        if (SPI1->SR & SPI_SR_RXNE) {
            uint8_t received = (uint8_t)(SPI1->DR & 0xFFU);
            (void)received;
        }
    }
}
