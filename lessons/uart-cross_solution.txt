#include "stm32f4xx_ll_bus.h"
#include "stm32f4xx_ll_gpio.h"
#include "stm32f4xx_ll_usart.h"

/*
 * External UART lesson:
 * Wire another UART device to PA9 (USART1_TX) and PA10 (USART1_RX).
 * Grounds must be shared.
 */
static void USART1_Init(void)
{
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA);
  LL_APB2_GRP1_EnableClock(LL_APB2_GRP1_PERIPH_USART1);

  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_9, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_8_15(GPIOA, LL_GPIO_PIN_9, LL_GPIO_AF_7);
  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_10, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_8_15(GPIOA, LL_GPIO_PIN_10, LL_GPIO_AF_7);

  LL_USART_SetBaudRate(USART1, 16000000U, 9600U);
  LL_USART_SetTransferDirection(USART1, LL_USART_DIRECTION_TX_RX);
  LL_USART_Enable(USART1);
}

static void USART1_WriteString(const char *text)
{
  while (*text != '\0')
  {
    while (!LL_USART_IsActiveFlag_TXE(USART1))
    {
    }
    LL_USART_TransmitData8(USART1, (uint8_t)(*text));
    text++;
  }
}

int main(void)
{
  USART1_Init();
  USART1_WriteString("USART1 ready\r\n");

  while (1)
  {
    if (LL_USART_IsActiveFlag_RXNE(USART1))
    {
      uint8_t received = LL_USART_ReceiveData8(USART1);
      while (!LL_USART_IsActiveFlag_TXE(USART1))
      {
      }
      LL_USART_TransmitData8(USART1, received);
    }
  }
}
