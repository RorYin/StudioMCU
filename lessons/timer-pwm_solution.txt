#include "stm32f446xe.h"

/*
 * TIM2 PWM on PA5 (AF1). LD2 brightness is controlled by CCR1 vs ARR.
 */
static void tim2_pwm_init(void) {
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;

    GPIOA->MODER &= ~GPIO_MODER_MODE5_Msk;
    GPIOA->MODER |= GPIO_MODER_MODE5_ALTERNATE;
    GPIOA->AFR[0] &= ~GPIO_AFR_AFSEL5_Msk;
    GPIOA->AFR[0] |= GPIO_AFR_AFSEL5_AF1;

    TIM2->PSC = 15U;          /* 16 MHz / 16 = 1 MHz timer clock */
    TIM2->ARR = 999U;         /* 1 kHz PWM */
    TIM2->CCR1 = 0U;
    TIM2->CCMR1 = TIM_CCMR1_OC1M_PWM1 | TIM_CCMR1_OC1PE;
    TIM2->CCER = TIM_CCER_CC1E;
    TIM2->CR1 = TIM_CR1_ARPE;
    TIM2->EGR = TIM_EGR_UG;
    TIM2->CR1 |= TIM_CR1_CEN;
}

int main(void) {
    tim2_pwm_init();
    TIM2->CCR1 = 500U; /* ~50% duty */
    while (1) { }
}
