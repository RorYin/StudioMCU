#include "stm32f4xx_ll_bus.h"
#include "stm32f4xx_ll_gpio.h"
#include "stm32f4xx_ll_tim.h"

/*
 * TIM2 PWM on PA5 (AF1). LD2 brightness is controlled by CCR1 vs ARR.
 */
static void TIM2_PWM_Init(void)
{
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA);
  LL_APB1_GRP1_EnableClock(LL_APB1_GRP1_PERIPH_TIM2);

  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_5, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_0_7(GPIOA, LL_GPIO_PIN_5, LL_GPIO_AF_1);

  LL_TIM_SetPrescaler(TIM2, 15U);       /* 16 MHz / 16 = 1 MHz */
  LL_TIM_SetAutoReload(TIM2, 999U);     /* 1 kHz PWM */
  LL_TIM_OC_SetCompareCH1(TIM2, 0U);
  LL_TIM_OC_SetMode(TIM2, LL_TIM_OCMODE_PWM1);
  LL_TIM_OC_EnablePreload(TIM2);
  LL_TIM_CC_EnableChannel(TIM2);
  LL_TIM_EnableARRPreload(TIM2);
  LL_TIM_GenerateEvent_UPDATE(TIM2);
  LL_TIM_EnableCounter(TIM2);
}

int main(void)
{
  TIM2_PWM_Init();
  LL_TIM_OC_SetCompareCH1(TIM2, 500U); /* ~50% duty */

  while (1)
  {
  }
}
