    // Enable GPIOA and USART2 clocks.

    // Set 115200 baud for the 16 MHz HSI clock (BRR = 0x008B).
    // Enable transmitter + USART (TE | UE).

static void uart2_write(const char *text) {
    while (*text) {
        while (!(USART2->SR & USART_SR_TXE)) { }
        USART2->DR = (uint32_t)(*text++);
    }
}

}
