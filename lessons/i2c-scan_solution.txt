#include "stm32f446xe.h"

/*
 * I2C1 on PB8=SCL, PB9=SDA (AF4).
 * Scan for any ACK on addresses 0x08..0x77 and report over USART2.
 */
static void uart2_init(void) {
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
    RCC->APB1ENR |= RCC_APB1ENR_USART2EN;
    GPIOA->MODER &= ~GPIO_MODER_MODE2_Msk;
    GPIOA->MODER |= GPIO_MODER_MODE2_ALTERNATE;
    GPIOA->AFR[0] &= ~GPIO_AFR_AFSEL2_Msk;
    GPIOA->AFR[0] |= GPIO_AFR_AFSEL2_AF7;
    USART2->BRR = 0x008BU;
    USART2->CR1 = USART_CR1_TE | USART_CR1_UE;
}

static void uart2_write(const char *text) {
    while (*text) {
        while (!(USART2->SR & USART_SR_TXE)) { }
        USART2->DR = (uint32_t)(*text++);
    }
}

static void i2c1_init(void) {
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOBEN;
    RCC->APB1ENR |= RCC_APB1ENR_I2C1EN;

    GPIOB->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOB->MODER |= (GPIO_MODER_MODE8_ALTERNATE | GPIO_MODER_MODE9_ALTERNATE);
    GPIOB->OTYPER |= (1U << 8) | (1U << 9); /* open-drain */
    GPIOB->PUPDR &= ~((3U << (8U * 2U)) | (3U << (9U * 2U)));
    GPIOB->PUPDR |= (1U << (8U * 2U)) | (1U << (9U * 2U)); /* pull-up */
    GPIOB->AFR[1] &= ~(GPIO_AFR_AFSEL8_Msk | GPIO_AFR_AFSEL9_Msk);
    GPIOB->AFR[1] |= (GPIO_AFR_AFSEL8_AF4 | GPIO_AFR_AFSEL9_AF4);

    I2C1->CR1 &= ~I2C_CR1_PE;
    I2C1->CR2 = 16U;     /* APB1 = 16 MHz */
    I2C1->CCR = 80U;     /* ~100 kHz */
    I2C1->TRISE = 17U;
    I2C1->CR1 |= I2C_CR1_PE;
}

static int i2c1_probe(uint8_t addr7) {
    I2C1->CR1 |= I2C_CR1_START;
    while (!(I2C1->SR1 & I2C_SR1_SB)) { }
    I2C1->DR = (uint32_t)(addr7 << 1);

    for (volatile uint32_t i = 0; i < 20000U; ++i) {
        if (I2C1->SR1 & I2C_SR1_ADDR) {
            (void)I2C1->SR1;
            (void)I2C1->SR2;
            I2C1->CR1 |= I2C_CR1_STOP;
            return 1;
        }
    }

    I2C1->CR1 |= I2C_CR1_STOP;
    return 0;
}

int main(void) {
    uart2_init();
    i2c1_init();
    uart2_write("I2C scan:\r\n");

    for (uint8_t addr = 0x08U; addr < 0x78U; ++addr) {
        if (i2c1_probe(addr)) {
            uart2_write("Found device\r\n");
        }
    }

    while (1) { }
}
