#include "stm32f4xx_ll_bus.h"
#include "stm32f4xx_ll_gpio.h"
#include "stm32f4xx_ll_spi.h"

/*
 * SPI loopback: short PA7 (MOSI) to PA6 (MISO) on the Nucleo headers.
 */
static void SPI1_Init(void)
{
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA);
  LL_APB2_GRP1_EnableClock(LL_APB2_GRP1_PERIPH_SPI1);

  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_5, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_0_7(GPIOA, LL_GPIO_PIN_5, LL_GPIO_AF_5);
  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_6, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_0_7(GPIOA, LL_GPIO_PIN_6, LL_GPIO_AF_5);
  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_7, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_0_7(GPIOA, LL_GPIO_PIN_7, LL_GPIO_AF_5);

  LL_SPI_SetMode(SPI1, LL_SPI_MODE_MASTER);
  LL_SPI_SetNSSModeSoft(SPI1);
  LL_SPI_Enable(SPI1);
}

static uint8_t SPI1_Transfer(uint8_t tx_byte)
{
  while (!LL_SPI_IsActiveFlag_TXE(SPI1))
  {
  }
  LL_SPI_TransmitData8(SPI1, tx_byte);

  while (!LL_SPI_IsActiveFlag_RXNE(SPI1))
  {
  }
  return LL_SPI_ReceiveData8(SPI1);
}

int main(void)
{
  SPI1_Init();

  while (1)
  {
    uint8_t response = SPI1_Transfer(0xA5U);
    (void)response;
  }
}
