#include "stm32f4xx_ll_bus.h"
#include "stm32f4xx_ll_gpio.h"
#include "stm32f4xx_ll_usart.h"
#include "stm32f4xx_ll_adc.h"
#include "stm32f4xx_ll_utils.h"

/*
 * Read ADC1 channel 0 on PA0 (Arduino A0).
 * Print the conversion result over USART2 (ST-Link VCP).
 */
static void USART2_Init(void)
{
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA);
  LL_APB1_GRP1_EnableClock(LL_APB1_GRP1_PERIPH_USART2);

  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_2, LL_GPIO_MODE_ALTERNATE);
  LL_GPIO_SetAFPin_0_7(GPIOA, LL_GPIO_PIN_2, LL_GPIO_AF_7);

  LL_USART_SetBaudRate(USART2, 16000000U, 115200U);
  LL_USART_SetTransferDirection(USART2, LL_USART_DIRECTION_TX);
  LL_USART_Enable(USART2);
}

static void USART2_WriteString(const char *text)
{
  while (*text != '\0')
  {
    while (!LL_USART_IsActiveFlag_TXE(USART2))
    {
    }
    LL_USART_TransmitData8(USART2, (uint8_t)(*text));
    text++;
  }
}

static void ADC1_Init(void)
{
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPIOA);
  LL_APB2_GRP1_EnableClock(LL_APB2_GRP1_PERIPH_ADC1);

  LL_GPIO_SetPinMode(GPIOA, LL_GPIO_PIN_0, LL_GPIO_MODE_ANALOG);
  LL_ADC_REG_SetSequencerRanks(ADC1, LL_ADC_CHANNEL_0);
  LL_ADC_Enable(ADC1);
}

static uint16_t ADC1_Read(void)
{
  LL_ADC_REG_StartConversionSWStart(ADC1);
  while (!LL_ADC_IsActiveFlag_EOCS(ADC1))
  {
  }
  return (uint16_t)LL_ADC_REG_ReadConversionData12(ADC1);
}

int main(void)
{
  char line[16];

  LL_Init1msTick(16000000U);
  USART2_Init();
  ADC1_Init();

  while (1)
  {
    uint16_t value = ADC1_Read();

    line[0] = 'A';
    line[1] = 'D';
    line[2] = 'C';
    line[3] = '=';
    line[4] = (char)('0' + ((value / 1000U) % 10U));
    line[5] = (char)('0' + ((value / 100U) % 10U));
    line[6] = (char)('0' + ((value / 10U) % 10U));
    line[7] = (char)('0' + (value % 10U));
    line[8] = '\r';
    line[9] = '\n';
    line[10] = '\0';

    USART2_WriteString(line);
    LL_mDelay(500);
  }
}
